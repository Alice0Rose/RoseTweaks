		}

			trigger = {

					has_valid_civic = civic_media_conglomerate

						has_valid_civic = civic_crowdsourcing

					}
				}
			}
			trade = @faction_improved_output
			mult = @media_conglomerate_multiplier
		}
		produces = {
			trigger = {
				owner = {
					OR = {
						has_valid_civic = civic_crowdsourcing
						has_valid_civic = civic_corporate_crowdsourcing
					}
				}
			}

			society_research = @faction_minor_output

		produces = {
			trigger = {
				owner = {
					NOR = {

						has_valid_civic = civic_corporate_crowdsourcing
					}
				}
			}
			physics_research = @faction_base_output
			mult = @faction_extortion_research_ratio
		}
		produces = {
			trigger = {
				owner = {
					OR = {
						has_valid_civic = civic_crowdsourcing
						has_valid_civic = civic_corporate_crowdsourcing
					}
				}
			}
			physics_research = @faction_minor_output
			society_research = @faction_minor_output
			engineering_research = @faction_minor_output
			mult = @faction_extortion_crowdsourcing_ratio
		}
	}

	#Note: number of potential or actual members is checked by POP_FACTION_MIN_POTENTIAL_MEMBERS define
	is_potential = {
		owner = {
			is_democratic_authority = yes
			has_country_flag = cyber_collectivist
		}
	}

	can_join_pre_triggers = {
		is_enslaved = no
		is_being_purged = no
	}

	can_join_faction = {
		custom_tooltip_icon = {
			fail_text = POP_CAN_THINK
			can_think = yes

		has_trait = trait_cybernetic

		base = 200

		base = 100

	demand = {
		title = "INTERLINK_NO_DEMANDS"
		unfulfilled_title = "INTERLINK_NO_DEMANDS"
		desc = "INTERLINK_NO_DEMANDS_DESC"

		fulfilled_effect = 0

		potential = {
			exists = owner
		}

			always = yes

				base = 0

			title = "EXTORT_FACTION"
			description = "EXTORT_FACTION_DESC"

			potential = {
				exists = owner
				owner = {
					is_oligarchic_authority = yes
					has_country_flag = cyber_collectivist
					OR = {
						has_valid_civic = civic_crowdsourcing

				exists = owner

					is_oligarchic_authority = yes
					has_country_flag = cyber_collectivist
				}
				is_faction_extorted = yes
			}

			effect = {
				set_faction_extorted = no
			}

			ai_weight = {
				base = 0
			}
		}
	}
}
