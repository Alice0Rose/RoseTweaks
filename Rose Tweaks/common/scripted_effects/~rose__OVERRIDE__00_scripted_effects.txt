give_next_terraforming_tech_option_effect = {
	if = {
		limit = { NOT = { has_technology = tech_terrestrial_sculpting } }
		add_research_option = tech_terrestrial_sculpting
		add_tech_progress = {
			tech = tech_terrestrial_sculpting
			progress = 0.25
		}
	}
	else_if = {
		limit = { NOT = { has_technology = tech_climate_restoration } }
		add_research_option = tech_climate_restoration
		add_tech_progress = {
			tech = tech_climate_restoration
			progress = 0.25
		}
	}
	else_if = {
		limit = { NOT = { has_technology = tech_ecological_adaptation } }
		add_research_option = tech_ecological_adaptation
		add_tech_progress = {
			tech = tech_ecological_adaptation
			progress = 0.25
		}
	}
	else = {
		add_monthly_resource_mult = {
			resource = society_research
			value = @tier3researchreward
			min = @tier3researchmin
			max = @tier3researchmax
		}
	}
}

# pop group scope. Moves AMOUNT pops into the assimilated species group
assimilate_pop_group_effect = {
	colony = {
		if = {
			limit = { root.owner = { is_gestalt = no } }
			create_pop_group = {
				pop_group = event_target:old_pop_group
				species = event_target:target_species
				size = 0
				category = civilian
				effect = {
					transfer_pop_amount = {
						source = event_target:old_pop_group
						target = this
						amount = $AMOUNT|5$
					}
					set_timed_pop_group_flag = {
						flag = already_assimilated
						days = 15
					}
					apply_post_assimilation_effects = yes
				}
			}
		}
		else = {
			create_pop_group = {
				pop_group = event_target:old_pop_group
				species = event_target:target_species
				size = 0
				category = maintenance_drone
				effect = {
					transfer_pop_amount = {
						source = event_target:old_pop_group
						target = this
						amount = $AMOUNT|5$
					}
					set_timed_pop_group_flag = {
						flag = already_assimilated
						days = 15
					}
					apply_post_assimilation_effects = yes
				}
			}
		}
	}
}
