namespace = cybernetics

### Create the Unemployed
### Planet scope gatekeeper
### on_building_complete, on_building_upgraded, on_building_demolished,
### on_building_downgraded, on_building_replaced
### on_district_complete, on_district_demolished
carrier_event = {
	id = cybernetics.2026
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		is_under_colonization = no
		NOT = { is_planet_class = pc_cosmogenesis_world }
		exists = owner
		owner = {
			is_fallen_empire = no
			has_active_tradition = tr_virtuality_finish
		}
		#free_jobs > 0
	}

	immediate = {
		carrier_event = { id = cybernetics.2027 days = 1 }
	}
}

### Create the Unemployed
### Fired by cybernetics.2025 and cybernetics.2026
carrier_event = {
	id = cybernetics.2027
	hide_window = yes

	is_triggered_only = yes
	trigger = {
		is_under_colonization = no
		NOT = { is_planet_class = pc_cosmogenesis_world }
		exists = owner
		owner = {
			is_fallen_empire = no
			has_tradition = tr_virtuality_finish
		}
		#free_jobs > 0
	}

	immediate = {
		set_variable = {
			which = pop_amount
			value = value:num_free_virtual_jobs
		}
		
		# Create pops for every free job
		if = {
			limit = {
				check_variable = {
					which = pop_amount
					value > 0
				}
			}
			
			create_pop_group = {
				species = owner_main_species
				size = pop_amount
			}
		}
		
		# Kill pops generated in excess (e.g. from forbidden jobs)
		if = {
			limit = {
				OR = {
					any_owned_pop_group = {
						is_unemployed = yes
						is_robotic_species = yes
						has_virtual_species_trait = yes
					}
					any_owned_pop_group = {
						is_robotic_species = yes
						has_virtual_species_trait = yes
						is_pop_category = civilian
					}
					any_owned_pop_group = {
						is_robotic_species = yes
						has_virtual_species_trait = yes
						is_pop_category = no_category
                        root = { is_under_colonization = no }
					}
				}
			}
			every_owned_pop_group = {
				limit = {
					OR = {
						AND = {
							is_unemployed = yes
							is_robotic_species = yes
							has_virtual_species_trait = yes
						}
						AND = {
							is_robotic_species = yes
							has_virtual_species_trait = yes
							is_pop_category = civilian
						}
						AND = {
							is_robotic_species = yes
							has_virtual_species_trait = yes
							is_pop_category = no_category
                            root = { is_under_colonization = no }
						}
					}
				}
				kill_pop_group = {
					pop_group = this
					percentage = 1
				}
			}
		}
		
		# Gets how many pops of all normal stratums there should be by checking the number of pops working each job	
		# Elite
		set_variable = {
			which = evpf_total_elite_jobs
			value = 0
		}
		
		every_owned_pop_job = {
			limit = { 
				has_job_category = ruler
			} 

			root = {
				change_variable = {
					which = evpf_total_elite_jobs
					value = prev.value:evpf_virtual_pops_assigned_to_job
				}
			}
		}
		
		# Specialist
		set_variable = {
			which = evpf_total_specialist_jobs
			value = 0
		}
		
		every_owned_pop_job = {
			limit = { 
				has_job_category = specialist
			}   
			
			root = {
				change_variable = {
					which = evpf_total_specialist_jobs
					value = prev.value:evpf_virtual_pops_assigned_to_job
				}
			}
		}
		
		# Worker
		set_variable = {
			which = evpf_total_worker_jobs
			value = 0
		}
		
		every_owned_pop_job = {
			limit = { 
				has_job_category = worker 
			} 
			root = {
				change_variable = {
					which = evpf_total_worker_jobs
					value = prev.value:evpf_virtual_pops_assigned_to_job
				}
			}
		}

		# Prevents removing all pops from a colony and decolonizing it by checking if there are no jobs,
		# and if so adding 1 to the job count of the highest stratum count with an existing pop group
		if = {
			limit = {
				check_variable = {
					which = evpf_total_elite_jobs
					value = 0
				}
				check_variable = {
					which = evpf_total_specialist_jobs
					value = 0
				}
				check_variable = {
					which = evpf_total_worker_jobs
					value = 0
				}
			}
			
			if = {
				limit = {
					count_owned_pop_group = {
						count > 0
						limit = { 
							is_pop_category = ruler
							has_virtual_species_trait = yes
						}
					}
				}
				change_variable = {
					which = evpf_total_elite_jobs
					value = 1
				}
			}
			else_if = {
				limit = {
					count_owned_pop_group = {
						count > 0
						limit = { 
							is_pop_category = specialist
							has_virtual_species_trait = yes
						}
					}
				}
				change_variable = {
					which = evpf_total_specialist_jobs
					value = 1
				}
			}
			else = {
				change_variable = {
					which = evpf_total_worker_jobs
					value = 1
				}
			}
		}

		# Get largest pop group of elite category
		ordered_owned_pop_group = {
			limit = { 
				is_pop_category = ruler 
				has_virtual_species_trait = yes
			}
			position = 0
			order_by = trigger:pop_group_size

			save_event_target_as = evpf_largest_pop_group_elite
		}
		
		if = {
			limit = {
				exists = event_target:evpf_largest_pop_group_elite
			}
			
			# Kill all other pop groups of elite category
			every_owned_pop_group = {
				limit = {
					is_pop_category = ruler
					has_virtual_species_trait = yes
					NOT = { is_same_value = event_target:evpf_largest_pop_group_elite }
				}
				
				kill_pop_group = {
					pop_group = this
					percentage = 1
				}
			}
			
			# Make largest pop group equal to evpf_total_elite_jobs
			event_target:evpf_largest_pop_group_elite = {
				# Add pops to largest group if size is less than
				if = {
					limit = {
						root = {
							# More jobs than pops, need more pops
							check_variable = { 
								which = evpf_total_elite_jobs
								value > prev.trigger:pop_group_size
							}
						}
					}
					
					set_variable = { 
						which = evpf_pops_to_add
						value = root.evpf_total_elite_jobs
					}
					
					subtract_variable = { 
						which = evpf_pops_to_add
						value = trigger:pop_group_size
					}
					
					add_pop_amount = evpf_pops_to_add
					clear_variable = evpf_pops_to_add
				}
				
				# Remove pops from largest group if size is greater than
				else_if = {
					limit = {
						# Less jobs than pops, need less pops
						root = {
							check_variable = { 
								which = evpf_total_elite_jobs
								value < prev.trigger:pop_group_size 
							}
						}
					}
				
					set_variable = { 
						which = evpf_pops_to_subtract
						value = trigger:pop_group_size
					}
				
					subtract_variable = { 
						which = evpf_pops_to_subtract
						value = root.evpf_total_elite_jobs
					}
					
					remove_pop_amount = evpf_pops_to_subtract
					clear_variable = evpf_pops_to_subtract
				}
			}
		}
		
		# Get largest pop group of specialist category
		ordered_owned_pop_group = {
			limit = { 
				is_pop_category = specialist 
				has_virtual_species_trait = yes
			}
			position = 0
			order_by = trigger:pop_group_size

			save_event_target_as = evpf_largest_pop_group_specialist
		}

		if = {
			limit = {
				exists = event_target:evpf_largest_pop_group_specialist
			}

			# Kill all other pop groups of specialist category
			every_owned_pop_group = {
				limit = {
					is_pop_category = specialist
					has_virtual_species_trait = yes
					NOT = { is_same_value = event_target:evpf_largest_pop_group_specialist }
				}
				
				kill_pop_group = {
					pop_group = this
					percentage = 1
				}
			}
			
			# Make largest pop group equal to evpf_total_specialist_jobs
			event_target:evpf_largest_pop_group_specialist = {
				# Add pops to largest group if size is less than
				if = {
					limit = {
						# More jobs than pops, need more pops
						root = {
							check_variable = { 
								which = evpf_total_specialist_jobs
								value > prev.trigger:pop_group_size
							}
						}
					}
					
					set_variable = { 
						which = evpf_pops_to_add
						value = root.evpf_total_specialist_jobs
					}
					
					subtract_variable = { 
						which = evpf_pops_to_add
						value = trigger:pop_group_size
					}
					
					add_pop_amount = evpf_pops_to_add
					clear_variable = evpf_pops_to_add
				}
				
				# Remove pops from largest group if size is greater than
				else_if = {
					limit = {
						# less jobs than pops, need less pops
						root = {
							check_variable = { 
								which = evpf_total_specialist_jobs
								value < prev.trigger:pop_group_size
							}
						}
					}
				
					set_variable = { 
						which = evpf_pops_to_subtract
						value = trigger:pop_group_size
					}
					
					subtract_variable = { 
						which = evpf_pops_to_subtract
						value = root.evpf_total_specialist_jobs
					}					
					
					remove_pop_amount = evpf_pops_to_subtract
					clear_variable = evpf_pops_to_subtract
				}
			}
		}
		
		# Get largest pop group of worker category
		ordered_owned_pop_group = {
			limit = { 
				is_pop_category = worker 
				has_virtual_species_trait = yes
			}
			position = 0
			order_by = trigger:pop_group_size

			save_event_target_as = evpf_largest_pop_group_worker
		}

		if = {
			limit = {
				exists = event_target:evpf_largest_pop_group_worker
			}
			
			# Kill all other pop groups of worker category
			every_owned_pop_group = {
				limit = {
					is_pop_category = worker
					has_virtual_species_trait = yes
					NOT = { is_same_value = event_target:evpf_largest_pop_group_worker }
				}
				
				kill_pop_group = {
					pop_group = this
					percentage = 1
				}
			}
			
			# Make largest pop group equal to evpf_total_worker_jobs
			event_target:evpf_largest_pop_group_worker = {
				# Add pops to largest group if size is less than
				if = {
					limit = {
						# more jobs than pops, need more pops
						root = {
							check_variable = { 
								which = evpf_total_worker_jobs
								value > prev.trigger:pop_group_size
							}
						}
					}
					
					set_variable = { 
						which = evpf_pops_to_add
						value = root.evpf_total_worker_jobs
					}
					
					subtract_variable = { 
						which = evpf_pops_to_add
						value = trigger:pop_group_size
					}
					
					add_pop_amount = evpf_pops_to_add
					clear_variable = evpf_pops_to_add
				}
				
				# Remove pops from largest group if size is greater than
				else_if = {
					limit = {
						# less jobs than pops, need less pops
						root = {
							check_variable = { 
								which = evpf_total_worker_jobs
								value < prev.trigger:pop_group_size
							}
						}
					}
				
					set_variable = { 
						which = evpf_pops_to_subtract
						value = trigger:pop_group_size
					}
					
					subtract_variable = { 
						which = evpf_pops_to_subtract
						value = root.evpf_total_worker_jobs
					}
					
					remove_pop_amount = evpf_pops_to_subtract
					clear_variable = evpf_pops_to_subtract
				}
			}
		}
		
		# Clean up used variables
		clear_variable = evpf_total_elite_jobs
		clear_variable = evpf_total_specialist_jobs
		clear_variable = evpf_total_worker_jobs
	}
}
